prediction_feedback Table (create this first)

CREATE TABLE prediction_feedback (
  id            UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
  zip_code      TEXT NOT NULL,
  metric        TEXT NOT NULL,       -- 'price', 'inventory', 'crime_trend', 'energy_load', 'traffic'
  model_type    TEXT NOT NULL,       -- 'prophet', 'scikit_learn'
  model_version TEXT NOT NULL,       -- 'v1.0', 'v1.1', etc.
  prediction_date TIMESTAMPTZ NOT NULL,
  predicted_value DECIMAL,
  actual_value    DECIMAL,           -- filled in when outcome is known
  confidence_pct  INTEGER,
  created_at    TIMESTAMPTZ DEFAULT NOW()
);
